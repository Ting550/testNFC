// Framework7 初始化（UI 專用） const app = new Framework7({ root: '#app' }); const allowedCardIds = ["ABC123456789"]; // 指定可接受的 NFC 卡號 document.getElementById("scanBtn").addEventListener("click", async () => { const statusBox = document.getElementById("statusBox"); const deviceInfo = document.getElementById("deviceInfo"); deviceInfo.style.display = "none"; statusBox.style.display = "block"; statusBox.className = "status-box"; statusBox.innerText = "請感應花博卡"; if (!("NDEFReader" in window)) { app.dialog.alert("此裝置不支援 Web NFC"); return; } try { const ndef = new NDEFReader(); await ndef.scan(); ndef.onreading = async event => { const cardId = event.serialNumber || ""; if (!allowedCardIds.includes(cardId)) { statusBox.classList.add("red"); statusBox.innerText = "非授權卡片"; return; } // 模擬呼叫 Controller API const apiResult = await mockApiCall(cardId); statusBox.className = "status-box"; if (apiResult === 1) { statusBox.classList.add("green"); statusBox.innerText = "通行成功"; } else if (apiResult === 2) { statusBox.classList.add("orange"); statusBox.innerText = "請稍候再試"; } else if (apiResult === 3) { statusBox.classList.add("red"); statusBox.innerText = "通行受限"; } else { statusBox.innerText = "未知狀態"; } }; } catch (err) { console.error(err); app.dialog.alert("NFC 啟用失敗"); } }); // 尚未串 API 前的模擬 function mockApiCall(cardId) { return new Promise(resolve => { setTimeout(() => { resolve(1); // 可測試 1 / 2 / 3 }, 1000); }); }
